{# @description One entry of a menu, used inside menu. Renders as a link when an href is given and as a button otherwise. An item with an icon and no text is named by an aria-label passed by the caller. The class and any attribute not listed below land on that link or button, where target, htmx attributes and a tooltip belong, and the list item around it takes none. #} {# @prop text:text | description:"The entry's label." #} {# @prop href:text | description:"Link address. Without it the entry is a button." #} {# @prop active:boolean | description:"Mark this entry as the current one." #} {# @prop disabled:boolean | description:"Show the entry as unavailable. A link loses its href and a button is disabled." #} {# @prop icon:text | description:"Icon shown before the text, passed as the name of the icon component." #} {# @prop class:text | description:"Extra classes added to the link or button." #} {# @prop type:select['button','submit','reset'] | description:"The button's type, for an entry without an href. With submit and a form attribute naming a form, the entry submits it." #} {# @prop aria-label:text | description:"Accessible name of the link or button, for an entry that shows only an icon." #} {# @slot — Extra content shown after the text. #} {% if href %} {% if icon %}{% endif %}{{ text }}{{ slot }} {% else %} {% endif %}
